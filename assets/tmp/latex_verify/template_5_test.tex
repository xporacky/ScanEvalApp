\newcommand{\ceilDiv}[1]{%
    \pgfmathtruncatemacro{\tempResult}{(#1 + 14)/15}%
    \setcounter{blocksNeeded}{\tempResult}%
}

\newcommand{\legendEmptyBox}[2]{%
    \draw[gray, thick] (#1,#2) rectangle ++(0.45,0.45);
}

\newcommand{\legendCrossBox}[2]{%
    \draw[gray, thick] (#1,#2) rectangle ++(0.45,0.45);
    \draw[thick] (#1+0.02,#2+0.02) -- (#1+0.43,#2+0.43);
    \draw[thick] (#1+0.02,#2+0.43) -- (#1+0.43,#2+0.02);
}

\newcommand{\legendCrossOutside}[2]{%
    \draw[gray, thick] (#1,#2) rectangle ++(0.45,0.45);
    \draw[thick] (#1+0.10,#2+0.20) -- (#1+0.50,#2+0.60);
    \draw[thick] (#1+0.10,#2+0.60) -- (#1+0.50,#2+0.20);
}

\newcommand{\legendFilledBox}[2]{%
    \draw[gray, thick] (#1,#2) rectangle ++(0.45,0.45);
    \fill[black!85] (#1,#2) rectangle (#1+0.45,#2+0.45);
}

\newcommand{\legendCircledFilledBox}[2]{%
    \draw[gray, thick] (#1,#2) rectangle ++(0.45,0.45);
    \fill[black!85] (#1,#2) rectangle (#1+0.45,#2+0.45);
    \draw[thick] (#1+0.225,#2+0.225) circle (0.31);
}

\newcommand{\generateBlock}[2]{%
    \setcounter{rowPosition}{1}

    \begin{tikzpicture}[font=\small]
        \draw[thick, black] (-1,-17) rectangle (11.5,-2);

        \foreach \position/\letter in {1/A,2/B,3/C,4/D,5/E} {
            \node at ({\position * 2}, -1.5) {\letter};
        }

        \foreach \line in {#1,...,#2} {
            \pgfmathsetmacro{\y}{-(\value{rowPosition}) - 1.5}

            \node at (0,\y) {\normalsize\textbf{\arabic{rowCounter}}};

            \foreach \position in {1,2,3,4,5} {
                \node[mygray] at ({\position * 2},\y) {\scalebox{2}{\textbf{$\times$}}};
                \node[draw, rectangle, inner sep=5pt] at ({\position * 2}, \y) {};
            }

            \ifnum\value{rowPosition}=5
                \draw[thin, black] (-1,-7.0) -- (11.5,-7.0);
            \fi
            \ifnum\value{rowPosition}=10
                \draw[thin, black] (-1,-12.0) -- (11.5,-12.0);
            \fi
            \ifnum\value{rowPosition}=15
                \draw[thin, black] (-1,-17.0) -- (11.5,-17.0);
            \fi

            \stepcounter{rowPosition}
            \stepcounter{rowCounter}
        }

        \node[anchor=west] at (-1,-17.5) {\footnotesize\textbf{Legenda:}};

        \node[anchor=west] at (-1,-18.2) {\footnotesize Správna odpoveď};
        \legendEmptyBox{-0.45}{-19.1}
        \legendEmptyBox{0.30}{-19.1}
        \legendCrossBox{1.05}{-19.1}
        \legendEmptyBox{1.80}{-19.1}

        \node[anchor=west] at (3.5,-18.2) {\footnotesize Nesprávna odpoveď};
        \legendCrossBox{4.05}{-19.1}
        \legendEmptyBox{4.80}{-19.1}
        \legendCrossBox{5.55}{-19.1}
        \legendEmptyBox{6.30}{-19.1}

        \node[anchor=west] at (8.0,-18.2) {\footnotesize Nesprávna odpoveď};
        \legendEmptyBox{8.55}{-19.1}
        \legendCrossOutside{9.30}{-19.1}
        \legendEmptyBox{10.05}{-19.1}
        \legendEmptyBox{10.80}{-19.1}

        \node[anchor=west] at (-1,-20.2) {\footnotesize Oprava};
        \legendCrossBox{-0.45}{-21.1}
        \legendEmptyBox{0.30}{-21.1}
        \legendFilledBox{1.05}{-21.1}
        \legendEmptyBox{1.80}{-21.1}

        \node[anchor=west] at (3.5,-20.2) {\footnotesize Spätná oprava};
        \legendCrossBox{4.05}{-21.1}
        \legendCrossBox{4.80}{-21.1}
        \legendCircledFilledBox{5.55}{-21.1}
        \legendCrossBox{6.30}{-21.1}
    \end{tikzpicture}%
}

\newcommand{\generateMultipleBlocks}[1]{%
    \ceilDiv{#1}
    \foreach \blockIndex in {1,...,\theblocksNeeded} {
        \pgfmathtruncatemacro{\startQuestion}{\value{rowCounter}}
        \pgfmathtruncatemacro{\endQuestion}{min(\value{rowCounter}+14,#1)}

        \generateBlock{\startQuestion}{\endQuestion}

        \ifnum\blockIndex<\theblocksNeeded\relax
            \newpage
        \fi
    }
}

\begin{document}
\begin{center}
    \generateMultipleBlocks{ 15 }
\end{center}
\end{document}
