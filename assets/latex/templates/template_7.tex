\newcounter{rowCounter}

\newcounter{rowPosition}
\newcounter{blocksNeeded}

\newcommand{\ceilDiv}[1]{%
    \pgfmathtruncatemacro{\tempResult}{(#1 + 19)/20}%
    \setcounter{blocksNeeded}{\tempResult}%
}

\newcommand{\generateBlock}[2]{%
    \setcounter{rowPosition}{1}

    \begin{tikzpicture}[font=\small]
        \draw[thick, black] (-1,-22) rectangle (15.5,-2);

        \foreach \position/\letter in {1/A,2/B,3/C,4/D,5/E,6/F,7/G} {
            \node at ({\position * 2}, -1.5) {\letter};
        }

        \foreach \line in {#1,...,#2} {
            \pgfmathsetmacro{\y}{-(\value{rowPosition}) - 1.5}

            \node at (0,\y) {\normalsize\textbf{\arabic{rowCounter}}};

            \foreach \position in {1,2,3,4,5,6,7} {
                \node[mygray] at ({\position * 2},\y) {\scalebox{2}{\textbf{$\times$}}};
                \node[draw, rectangle, minimum size=0.45cm, inner sep=0pt] at ({\position * 2}, \y) {};
            }

            \ifnum\value{rowPosition}=5
                \draw[thin, black] (-1,-7.0) -- (15.5,-7.0);
            \fi
            \ifnum\value{rowPosition}=10
                \draw[thin, black] (-1,-12.0) -- (15.5,-12.0);
            \fi
            \ifnum\value{rowPosition}=15
                \draw[thin, black] (-1,-17.0) -- (15.5,-17.0);
            \fi
            \ifnum\value{rowPosition}=20
                \draw[thin, black] (-1,-22.0) -- (15.5,-22.0);
            \fi

            \stepcounter{rowPosition}
            \stepcounter{rowCounter}
        }

    \end{tikzpicture}%
}

\newcommand{\generateMultipleBlocks}[1]{%
    \ceilDiv{#1}
    \foreach \blockIndex in {1,...,\theblocksNeeded} {
        \pgfmathtruncatemacro{\startQuestion}{\value{rowCounter}}
        \pgfmathtruncatemacro{\endQuestion}{min(\value{rowCounter}+19,#1)}

        \generateBlock{\startQuestion}{\endQuestion}

        \ifnum\blockIndex<\theblocksNeeded\relax
            \newpage
        \fi
    }
}
